NAVY = "#0B2F5B"
LIGHT_BLUE = "#DCE7F5"
TAB_ACCENTS = ("#2878B5", "#169C8C", "#D18B22")


TAB_GROUPS = (
    (
        "Manipulating/Combinable/BF Files",
        (MANUAL, GENERIC_CONTINGENCY),
    ),
    (
        "BF/P2+Switching Files",

    (
        "Non-Combinable/Op Guide Files",

SECTION_LABELS = {
    MANUAL: "Manipulating AUX",
    GENERIC_CONTINGENCY: "Generic and External Contingency AUX/CON applied BEFORE Join Active AUX (Combinable Conts)",
    JOIN_ACTIVE_CONTINGENCY: "Generic Contingency AUX/CON applied AFTER Join Active AUX (BF/Non-Combinable/P2+Switching Conts)",
    OPERATING_GUIDE: "Operating Guide Contingency AUX",
    GENERIC_CONTINGENCY_POST: "External Conts (Non-Combinable)",
    COMMON_STRUCTURE: "Common Structure Conts (Non-Combinable)",
}


class _CategoryView(ttk.LabelFrame):
    def __init__(self, parent: tk.Misc, category: str, cases: list[str]) -> None:
        super().__init__(parent, text=SECTION_LABELS[category], padding=8)
        self.category = category
        self.cases = cases
        self.entries: list[AuxFileEntry] = []
        self._checkbox_variables: list[tk.BooleanVar] = []

        self.canvas = tk.Canvas(
            self,
            background="#F7F9FC",
            highlightthickness=1,
            highlightbackground="#B8C9DC",
            height=120,
        )
        self.table = ttk.Frame(self.canvas, padding=(4, 2))
        self._table_window = self.canvas.create_window((0, 0), window=self.table, anchor="nw")
        vertical = ttk.Scrollbar(self, orient="vertical", command=self.canvas.yview)
        horizontal = ttk.Scrollbar(self, orient="horizontal", command=self.canvas.xview)
        self.canvas.configure(yscrollcommand=vertical.set, xscrollcommand=horizontal.set)
        self.table.bind("<Configure>", self._table_configured)
        self.canvas.grid(row=0, column=0, columnspan=3, sticky="nsew")
        vertical.grid(row=0, column=3, sticky="ns")
        horizontal.grid(row=1, column=0, columnspan=3, sticky="ew")
        self.rowconfigure(0, weight=1)
        self.columnconfigure(0, weight=1)

        ttk.Button(self, text="Select Files...", command=self._add).grid(row=2, column=0, sticky="w", pady=(8, 0))

    def _table_configured(self, _event=None) -> None:
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))

    def _case_heading(self, case: str, index: int) -> str:
        name = case.replace("/", "\\").rstrip("\\").split("\\")[-1]
        if sum(
            other.replace("/", "\\").rstrip("\\").split("\\")[-1].casefold()
            == name.casefold()
            for other in self.cases
        ) > 1:
            return f"{name} ({index + 1})"
        return name

    def set_records(self, records: list[dict[str, object]]) -> None:
        self.entries = decode_entries(records)
        self._refresh()

    def records(self) -> list[dict[str, object]]:
        return encode_entries(self.entries)

    def _refresh(self) -> None:
        for child in self.table.winfo_children():
            child.destroy()
        self._checkbox_variables.clear()

        headers = ("File Name", "File Exists", "All cases") + tuple(
            self._case_heading(case, index) for index, case in enumerate(self.cases)
        ) + ("",)
        for column, text in enumerate(headers):
            ttk.Label(
                self.table,
                text=text,
                font=("Segoe UI", 9, "bold"),
                anchor="center" if column else "w",
            ).grid(row=0, column=column, sticky="ew", padx=6, pady=(3, 5))

        for index, entry in enumerate(self.entries):
            row = index + 1
            ttk.Label(self.table, text=entry.path, width=68, anchor="w").grid(
                row=row, column=0, sticky="ew", padx=6, pady=3
            )
            ttk.Label(
                self.table,
                text="Yes" if entry.exists else "No",
                anchor="center",
            ).grid(row=row, column=1, sticky="ew", padx=6, pady=3)

            all_var = tk.BooleanVar(value=entry.apply_to_all)
            self._checkbox_variables.append(all_var)
            ttk.Checkbutton(
                self.table,
                variable=all_var,
                command=lambda item=index, value=all_var: self._set_all_cases(item, value),
            ).grid(row=row, column=2, padx=22, pady=3)

            for case_index, case in enumerate(self.cases):
                case_var = tk.BooleanVar(value=entry.applies_to(case))
                self._checkbox_variables.append(case_var)
                ttk.Checkbutton(
                    self.table,
                    variable=case_var,
                    command=lambda item=index, selected_case=case, value=case_var: (
                        self._set_case(item, selected_case, value)
                    ),
                ).grid(row=row, column=case_index + 3, padx=22, pady=3)

            ttk.Button(
                self.table,
                text="Remove",
                command=lambda item=index: self._remove(item),
            ).grid(row=row, column=len(self.cases) + 3, padx=6, pady=3)

        self.table.columnconfigure(0, weight=1, minsize=430)
        self.table.update_idletasks()
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))

    def _set_all_cases(self, index: int, variable: tk.BooleanVar) -> None:
        self.entries[index].set_all_cases(variable.get())
        self._refresh()

    def _set_case(
        self,
        index: int,
        case: str,
        variable: tk.BooleanVar,
    ) -> None:
        self.entries[index].set_case_enabled(case, variable.get(), self.cases)
        self._refresh()

    def _add(self) -> None:
        paths = filedialog.askopenfilenames(
            parent=self,
            title=f"Add {CATEGORY_LABELS[self.category]} files",
            filetypes=(("PowerWorld AUX/CON", "*.aux *.con"), ("All files", "*.*")),
        )
        known = {entry.path.casefold() for entry in self.entries}
        for path in paths:
            if path.casefold() not in known:
                self.entries.append(AuxFileEntry(path=path))
                known.add(path.casefold())
        self._refresh()

    def _remove(self, index: int) -> None:
        del self.entries[index]
        self._refresh()

class AuxSelectionDialog(tk.Toplevel):
    def __init__(self, parent: tk.Misc, settings: AppSettings, store: SettingsStore, cases: list[str], on_saved=None) -> None:
        super().__init__(parent)
        self.settings = settings
        self.store = store
        self.cases = cases
        self.on_saved = on_saved
        self.title("Select AUX Files")
        screen_width = self.winfo_screenwidth()
        screen_height = self.winfo_screenheight()
        window_width = min(max(1080, int(screen_width * 0.72)), screen_width - 50)
        window_height = min(max(760, int(screen_height * 0.88)), screen_height - 80)
        left = max(20, (screen_width - window_width) // 2)
        top = max(20, (screen_height - window_height) // 2)
        self.geometry(f"{window_width}x{window_height}+{left}+{top}")
        self.minsize(min(900, screen_width - 40), min(620, screen_height - 70))
        self.transient(parent)
        self.grab_set()

        self.rowconfigure(1, weight=1)
        self.columnconfigure(0, weight=1)

        header = tk.Frame(self, bg=NAVY)
        header.grid(row=0, column=0, sticky="ew")
        tk.Label(
            header,
            text="Select AUX Files",
            bg=NAVY,
            fg="white",
            font=("Segoe UI", 15, "bold"),
            padx=16,
            pady=10,
        ).pack(side="left")
        tk.Label(
            header,
            text="Assign each AUX/CON file to all cases or selected cases",
            bg=NAVY,
            fg=LIGHT_BLUE,
            font=("Segoe UI", 10),
            padx=8,
        ).pack(side="left")

        body = ttk.Frame(self, padding=12)
        body.grid(row=1, column=0, sticky="nsew")
        body.rowconfigure(1, weight=1)
        body.columnconfigure(0, weight=1)
        ttk.Label(
            body,
            text=(
                "Choose AUX/CON files, then click All cases or an individual case checkbox. "
                "Each case column shows the case filename."
            ),
        ).grid(row=0, column=0, sticky="w", pady=(0, 8))

        style = ttk.Style(self)
        style.configure(
            "Aux.TNotebook",
            background=LIGHT_BLUE,
            borderwidth=0,
            tabmargins=(2, 2, 2, 0),
        )
        style.configure(
            "Aux.TNotebook.Tab",
            background="#C8D8EB",
            foreground="#17324D",
            font=("Segoe UI", 10, "bold"),
            padding=(14, 8),
        )
        style.map(
            "Aux.TNotebook.Tab",
            background=[("selected", NAVY), ("active", "#AFC8E3")],
            foreground=[("selected", "white"), ("active", "#0B2F5B")],
        )

        notebook = ttk.Notebook(body, style="Aux.TNotebook")
        notebook.grid(row=1, column=0, sticky="nsew")
        self._tab_icons: list[tk.PhotoImage] = []
        self.views: dict[str, _CategoryView] = {}
        for tab_index, (tab_name, categories) in enumerate(TAB_GROUPS):
            tab = ttk.Frame(notebook, padding=8)
            accent = TAB_ACCENTS[tab_index]
            icon = tk.PhotoImage(width=12, height=12)
            icon.put(accent, to=(0, 0, 12, 12))
            self._tab_icons.append(icon)
            notebook.add(tab, text=tab_name, image=icon, compound="left")
            tk.Frame(tab, bg=accent, height=5).pack(fill="x", pady=(0, 7))
            for category in categories:
                view = _CategoryView(tab, category, cases)
                view.pack(fill="both", expand=True, pady=(0, 7))
                view.set_records(settings.aux_files.get(category, []))
                self.views[category] = view

        buttons = ttk.Frame(body)
        buttons.grid(row=2, column=0, sticky="ew", pady=(10, 0))
        ttk.Button(
            buttons,
            text="Configure Computers and Cores...",
            command=self._open_distributed,
        ).pack(side="left")
        ttk.Button(buttons, text="Load Defaults", command=self._load_defaults).pack(side="left", padx=8)
        ttk.Button(buttons, text="Cancel", command=self.destroy).pack(side="right")
        ttk.Button(buttons, text="Save", command=self._save).pack(side="right", padx=8)

    def _open_distributed(self) -> None:
        DistributedDialog(self, self.settings, self.store)

    def _load_defaults(self) -> None:
        if not messagebox.askyesno("AUX defaults", "Replace current AUX selections with the legacy defaults?", parent=self):
            return
        defaults = default_aux_selections(self.cases)
        for category, view in self.views.items():
            view.set_records(defaults[category])

    def _save(self) -> None:
        selections = {category: view.records() for category, view in self.views.items()}
        self.settings.aux_files = selections
        self.store.save(self.settings)
        if self.on_saved is not None:
            self.on_saved(sum(len(records) for records in selections.values()))
        self.destroy()
