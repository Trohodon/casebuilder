"""Distributed-computing configuration and contingency batching."""

from __future__ import annotations

from dataclasses import dataclass
import os
from pathlib import Path
import platform
import socket
import subprocess
from typing import Iterator, Sequence


@dataclass(frozen=True, slots=True)
class DistributedComputer:
    name: str
    connections: int

    def validate(self) -> None:
        name = self.name.strip()
        if not name:
            raise ValueError("Every distributed computer needs a name.")
        if any(character in name for character in ('"', "\r", "\n")):
            raise ValueError(f"Invalid computer name: {self.name!r}")
        if self.connections < 1:
            raise ValueError(f"Cores/workers for {name} must be at least 1.")


@dataclass(frozen=True, slots=True)
class ComputerVerification:
    reachable: bool | None
    message: str


def verify_computer(
    name: str,
    timeout_seconds: float = 2.0,
    *,
    runner=subprocess.run,
    system_name: str | None = None,
) -> ComputerVerification:
    """Do a quick reachability check without requiring PowerWorld or SimAuto."""
    computer = name.strip()
    if not computer:
        return ComputerVerification(None, "Blank - ignored when running")

    local_names = {
        "localhost",
        ".",
        socket.gethostname().casefold(),
        socket.getfqdn().casefold(),
        os.environ.get("COMPUTERNAME", "").casefold(),
    }
    if computer.casefold() in local_names:
        return ComputerVerification(True, "Ready - this computer")

    current_system = system_name or platform.system()
    milliseconds = max(250, int(timeout_seconds * 1000))
    if current_system == "Windows":
        command = ["ping", "-n", "1", "-w", str(milliseconds), computer]
    else:
        command = ["ping", "-c", "1", "-W", str(max(1, round(timeout_seconds))), computer]
    try:
        completed = runner(
            command,
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            timeout=timeout_seconds + 1,
            check=False,
        )
    except (OSError, subprocess.SubprocessError):
        return ComputerVerification(False, "Could not run network check")
    if completed.returncode == 0:
        return ComputerVerification(True, "Reachable")
    return ComputerVerification(False, "No reply - offline or ping is blocked")


def parse_powerworld_verification(result: object) -> dict[str, ComputerVerification]:
    """Translate SimAuto's distributed-computer table into GUI status messages."""
    try:
        columns = result[1]  # type: ignore[index]
        names, availability, core_counts, error_counts = columns
        rows = zip(names, availability, core_counts, error_counts)
    except (IndexError, TypeError, ValueError) as exc:
        raise ValueError("PowerWorld returned an unexpected distributed-computer list.") from exc

    verified: dict[str, ComputerVerification] = {}
    for name, available, cores, errors in rows:
        computer = str(name).strip()
        is_available = str(available).strip().casefold() in {"yes", "true", "1"}
        if is_available:
            try:
                detected = int(float(str(cores)))
            except ValueError:
                detected = 0
            detail = f"PowerWorld ready - {detected} core(s) found" if detected else "PowerWorld ready"
        else:
            error_text = str(errors).strip()
            detail = "PowerWorld unavailable"
            if error_text and error_text != "0":
                detail += f" - {error_text} error(s)"
        verified[computer.casefold()] = ComputerVerification(is_available, detail)
    return verified


@dataclass(frozen=True, slots=True)
class DistributedConfig:
    contingencies_per_process: int
    computers: tuple[DistributedComputer, ...]

    def validate(self) -> None:
        if self.contingencies_per_process < 1:
            raise ValueError("Contingencies per process must be at least 1.")
        if not self.computers:
            raise ValueError("At least one distributed computer is required.")
        for computer in self.computers:
            computer.validate()


def render_distributed_aux(config: DistributedConfig) -> str:
    """Build PowerWorld's distributed-options AUX text without using UI labels."""
    config.validate()

    lines = [
        "DATA (DISTRIBUTED_OPTIONS, [ATCUseDistributedComputing,CTGUseDistributedComputing,DistMasterPasswordHash,",
        "    CTGNumberPerProcess])",
        "{",
        f'"NO " "YES" "" {config.contingencies_per_process}',
        "}",
        "",
        "//--------------------------------------------------------------------------------",
        "// THE FOLLOWING SECTION CONTAINS OPTIONS FOR DISTRIBUTED COMPUTING.",
        "//--------------------------------------------------------------------------------",
        "DATA (DISTRIBUTEDCOMPUTER, [CustomExpression,CustomExpression:1,Selected,NumberOfConnections,",
        "    NumberOfConnections:1,DSServerName,Enabled,Enabled:1,DistNumErrors,",
        "    DistMaxErrors,DistAuthInfoStored,DistAuthInfo,DistAuthInfo:1,DistAuthInfo:2,",
        "    CustomExpressionStr,CustomExpressionStr:1])",
        "{",
    ]
    for computer in config.computers:
        lines.append(
            f'"" "" "NO " {computer.connections} 0 "{computer.name.strip()}" '
            '"YES" "Unknown" 0 1 "NO " "" "" "" "" ""'
        )
    lines.extend(("}", ""))
    return "\n".join(lines)


def write_distributed_aux(path: Path, config: DistributedConfig) -> Path:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(render_distributed_aux(config), encoding="utf-8", newline="\n")
    return path


def contingency_batch_bounds(total: int, batch_size: int) -> Iterator[tuple[int, int]]:
    """Yield zero-based [start, end) bounds without dropping boundary items."""
    if total < 0:
        raise ValueError("Total contingencies cannot be negative.")
    if batch_size < 1:
        raise ValueError("Batch size must be at least 1.")
    for start in range(0, total, batch_size):
        yield start, min(start + batch_size, total)


def split_sequence(values: Sequence[str], batch_size: int) -> list[list[str]]:
    return [list(values[start:end]) for start, end in contingency_batch_bounds(len(values), batch_size)]
