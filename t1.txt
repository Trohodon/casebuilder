from __future__ import annotations

import tkinter as tk
from tkinter import filedialog, messagebox, ttk

from core.aux_files import (
    CATEGORY_LABELS,
    COMMON_STRUCTURE,
    GENERIC_CONTINGENCY,
    GENERIC_CONTINGENCY_POST,
    JOIN_ACTIVE_CONTINGENCY,
    MANUAL,
    OPERATING_GUIDE,
    AuxFileEntry,
    decode_entries,
    default_aux_selections,
    encode_entries,
)
from core.settings import AppSettings, SettingsStore
from .distributed_dialog import DistributedDialog


TAB_GROUPS = (
    (
        "Manipulating/Combinable/BF Files",
        (MANUAL, GENERIC_CONTINGENCY),
    ),
    (
        "BF/P2+Switching Files",
        (JOIN_ACTIVE_CONTINGENCY,),
    ),
    (
        "Non-Combinable/Op Guide Files",
        (OPERATING_GUIDE, GENERIC_CONTINGENCY_POST, COMMON_STRUCTURE),
    ),
)

SECTION_LABELS = {
    MANUAL: "Manipulating AUX",
    GENERIC_CONTINGENCY: "Generic and External Contingency AUX/CON applied BEFORE Join Active AUX (Combinable Conts)",
    JOIN_ACTIVE_CONTINGENCY: "Generic Contingency AUX/CON applied AFTER Join Active AUX (BF/Non-Combinable/P2+Switching Conts)",
    OPERATING_GUIDE: "Operating Guide Contingency AUX",
    GENERIC_CONTINGENCY_POST: "External Conts (Non-Combinable)",
    COMMON_STRUCTURE: "Common Structure Conts (Non-Combinable)",
}


class _CategoryView(ttk.LabelFrame):
    def __init__(self, parent: tk.Misc, category: str, cases: list[str]) -> None:
        super().__init__(parent, text=SECTION_LABELS[category], padding=8)
        self.category = category
        self.cases = cases
        self.entries: list[AuxFileEntry] = []

        self.tree = ttk.Treeview(
            self,
            columns=("path", "exists", "cases"),
            show="headings",
            selectmode="extended",
            height=4,
        )
        self.tree.heading("path", text="File Name")
        self.tree.heading("exists", text="File Exists")
        self.tree.heading("cases", text="Use For Case")
        self.tree.column("path", width=610, anchor="w")
        self.tree.column("exists", width=70, anchor="center")
        self.tree.column("cases", width=110, anchor="center")
        scrollbar = ttk.Scrollbar(self, orient="vertical", command=self.tree.yview)
        self.tree.configure(yscrollcommand=scrollbar.set)
        self.tree.grid(row=0, column=0, columnspan=5, sticky="nsew")
        scrollbar.grid(row=0, column=5, sticky="ns")
        self.rowconfigure(0, weight=1)
        self.columnconfigure(0, weight=1)

        ttk.Button(self, text="Select Files...", command=self._add).grid(row=1, column=0, sticky="w", pady=(8, 0))
        ttk.Button(self, text="Remove", command=self._remove).grid(row=1, column=1, sticky="w", padx=6, pady=(8, 0))
        ttk.Button(self, text="Use for all cases", command=self._use_all).grid(row=1, column=2, sticky="w", pady=(8, 0))
        ttk.Button(self, text="Choose cases...", command=self._choose_cases).grid(row=1, column=3, sticky="w", padx=6, pady=(8, 0))

    def set_records(self, records: list[dict[str, object]]) -> None:
        self.entries = decode_entries(records)
        self._refresh()

    def records(self) -> list[dict[str, object]]:
        return encode_entries(self.entries)

    def _refresh(self) -> None:
        self.tree.delete(*self.tree.get_children())
        for index, entry in enumerate(self.entries):
            usage = "All cases" if entry.apply_to_all else f"{len(entry.cases)} selected"
            self.tree.insert("", "end", iid=str(index), values=(entry.path, "Yes" if entry.exists else "No", usage))

    def _selected_indexes(self) -> list[int]:
        return sorted((int(item) for item in self.tree.selection()), reverse=True)

    def _add(self) -> None:
        paths = filedialog.askopenfilenames(
            parent=self,
            title=f"Add {CATEGORY_LABELS[self.category]} files",
            filetypes=(("PowerWorld AUX/CON", "*.aux *.con"), ("All files", "*.*")),
        )
        known = {entry.path.casefold() for entry in self.entries}
        for path in paths:
            if path.casefold() not in known:
                self.entries.append(AuxFileEntry(path=path))
                known.add(path.casefold())
        self._refresh()

    def _remove(self) -> None:
        for index in self._selected_indexes():
            del self.entries[index]
        self._refresh()

    def _use_all(self) -> None:
        for index in self._selected_indexes():
            self.entries[index].apply_to_all = True
            self.entries[index].cases = []
        self._refresh()

    def _choose_cases(self) -> None:
        selected = self._selected_indexes()
        if not selected:
            messagebox.showinfo("AUX files", "Select at least one AUX row first.", parent=self)
            return
        if not self.cases:
            messagebox.showinfo("AUX files", "Add PowerWorld cases on the main screen first.", parent=self)
            return
        dialog = tk.Toplevel(self)
        dialog.title("Choose cases")
        dialog.geometry("650x360")
        dialog.transient(self.winfo_toplevel())
        dialog.grab_set()
        ttk.Label(dialog, text="Use the selected AUX file(s) for these cases:").pack(anchor="w", padx=12, pady=(12, 4))
        case_list = tk.Listbox(dialog, selectmode="extended")
        case_list.pack(fill="both", expand=True, padx=12, pady=4)
        for case in self.cases:
            case_list.insert("end", case)

        def apply() -> None:
            chosen = [self.cases[index] for index in case_list.curselection()]
            for index in selected:
                self.entries[index].apply_to_all = False
                self.entries[index].cases = chosen
            self._refresh()
            dialog.destroy()

        ttk.Button(dialog, text="Apply", command=apply).pack(anchor="e", padx=12, pady=12)


class AuxSelectionDialog(tk.Toplevel):
    def __init__(self, parent: tk.Misc, settings: AppSettings, store: SettingsStore, cases: list[str], on_saved=None) -> None:
        super().__init__(parent)
        self.settings = settings
        self.store = store
        self.cases = cases
        self.on_saved = on_saved
        self.title("Select AUX Files")
        self.geometry("1080x760")
        self.minsize(900, 620)
        self.transient(parent)
        self.grab_set()

        body = ttk.Frame(self, padding=12)
        body.pack(fill="both", expand=True)
        ttk.Label(
            body,
            text="Choose AUX/CON files and the PowerWorld cases that should use them.",
        ).pack(anchor="w", pady=(0, 8))
        notebook = ttk.Notebook(body)
        notebook.pack(fill="both", expand=True)
        self.views: dict[str, _CategoryView] = {}
        for tab_name, categories in TAB_GROUPS:
            tab = ttk.Frame(notebook, padding=8)
            notebook.add(tab, text=tab_name)
            for category in categories:
                view = _CategoryView(tab, category, cases)
                view.pack(fill="both", expand=True, pady=(0, 7))
                view.set_records(settings.aux_files.get(category, []))
                self.views[category] = view

        buttons = ttk.Frame(body)
        buttons.pack(fill="x", pady=(10, 0))
        ttk.Button(
            buttons,
            text="Configure Computers and Cores...",
            command=self._open_distributed,
        ).pack(side="left")
        ttk.Button(buttons, text="Load Defaults", command=self._load_defaults).pack(side="left", padx=8)
        ttk.Button(buttons, text="Cancel", command=self.destroy).pack(side="right")
        ttk.Button(buttons, text="Save", command=self._save).pack(side="right", padx=8)

    def _open_distributed(self) -> None:
        DistributedDialog(self, self.settings, self.store)

    def _load_defaults(self) -> None:
        if not messagebox.askyesno("AUX defaults", "Replace current AUX selections with the legacy defaults?", parent=self):
            return
        defaults = default_aux_selections(self.cases)
        for category, view in self.views.items():
            view.set_records(defaults[category])

    def _save(self) -> None:
        selections = {category: view.records() for category, view in self.views.items()}
        self.settings.aux_files = selections
        self.store.save(self.settings)
        if self.on_saved is not None:
            self.on_saved(sum(len(records) for records in selections.values()))
        self.destroy()
