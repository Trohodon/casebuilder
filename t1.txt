import unittest
from unittest.mock import Mock

from core.distributed import (
    DistributedComputer,
    DistributedConfig,
    contingency_batch_bounds,
    parse_powerworld_verification,
    render_distributed_aux,
    split_sequence,
    verify_computer,
)


class DistributedConfigTests(unittest.TestCase):
    def test_aux_contains_each_computer_once(self) -> None:
        config = DistributedConfig(
            100,
            (
                DistributedComputer("localhost", 4),
                DistributedComputer("S99589", 2),
            ),
        )
        text = render_distributed_aux(config)
        self.assertIn('"NO " "YES" "" 100', text)
        self.assertEqual(text.count('"localhost"'), 1)
        self.assertEqual(text.count('"S99589"'), 1)

    def test_invalid_connections_are_rejected(self) -> None:
        config = DistributedConfig(100, (DistributedComputer("server", 0),))
        with self.assertRaises(ValueError):
            render_distributed_aux(config)

    def test_blank_computer_is_ignored_by_verification(self) -> None:
        result = verify_computer("   ")
        self.assertIsNone(result.reachable)
        self.assertIn("ignored", result.message)

    def test_remote_computer_reachability_uses_ping_result(self) -> None:
        runner = Mock(return_value=Mock(returncode=0))
        result = verify_computer("planning-worker", runner=runner, system_name="Windows")
        self.assertTrue(result.reachable)
        self.assertEqual(runner.call_args.args[0][:3], ["ping", "-n", "1"])

    def test_powerworld_verification_reports_availability_and_detected_cores(self) -> None:
        result = ("", [["localhost", "WORKER1"], ["YES", "NO"], ["16", "8"], ["0", "2"]])
        statuses = parse_powerworld_verification(result)
        self.assertTrue(statuses["localhost"].reachable)
        self.assertIn("16 core", statuses["localhost"].message)
        self.assertFalse(statuses["worker1"].reachable)
        self.assertIn("2 error", statuses["worker1"].message)

    def test_batch_bounds_are_zero_based_and_complete(self) -> None:
        self.assertEqual(list(contingency_batch_bounds(7, 3)), [(0, 3), (3, 6), (6, 7)])
        values = [str(index) for index in range(7)]
        self.assertEqual(split_sequence(values, 3), [["0", "1", "2"], ["3", "4", "5"], ["6"]])


if __name__ == "__main__":
    unittest.main()
