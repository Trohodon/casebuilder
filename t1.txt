@echo off
setlocal

set "REPO_ROOT=%~dp0.."
set "PUBLISH_DIR=%REPO_ROOT%\dist\NERC TPL CA"
set "APP_EXE=%PUBLISH_DIR%\NERC TPL CA.exe"
set "ICON_PATH=%REPO_ROOT%\NERC_TPL_CA_PY\assets\app.ico"
set "OUTPUT_DIR=%~dp0output"
set "PACKAGE_SOURCE=%~dp0Package.wxs"
set "VERSION=%~1"

if not defined VERSION set "VERSION=1.0.0"

if not exist "%APP_EXE%" (
    echo ERROR: The PyInstaller application was not found:
    echo   %APP_EXE%
    echo.
    echo Build the onedir EXE first, then run this script again.
    exit /b 1
)

if not exist "%ICON_PATH%" (
    echo ERROR: The installer icon was not found:
    echo   %ICON_PATH%
    exit /b 1
)

where wix.exe >nul 2>nul
if %errorlevel% equ 0 (
    set "WIX=wix.exe"
) else if exist "%USERPROFILE%\.dotnet\tools\wix.exe" (
    set "WIX=%USERPROFILE%\.dotnet\tools\wix.exe"
) else (
    echo ERROR: WiX was not found in PATH or %USERPROFILE%\.dotnet\tools.
    exit /b 1
)

if not exist "%OUTPUT_DIR%" mkdir "%OUTPUT_DIR%"

"%WIX%" extension list | findstr /I /C:"WixToolset.UI.wixext 7.0.0" >nul
if errorlevel 1 (
    echo ERROR: The WiX 7 UI extension is not installed.
    echo Run this once, then run this build again:
    echo   wix extension add WixToolset.UI.wixext/7.0.0
    exit /b 1
)

echo Building NERC TPL CA version %VERSION%...
"%WIX%" build ^
    -arch x64 ^
    -ext WixToolset.UI.wixext ^
    -culture en-US ^
    -d "Version=%VERSION%" ^
    -d "PublishDir=%PUBLISH_DIR%" ^
    -d "IconPath=%ICON_PATH%" ^
    -out "%OUTPUT_DIR%\NERC_TPL_CA_%VERSION%_x64.msi" ^
    "%PACKAGE_SOURCE%"

if errorlevel 1 (
    echo.
    echo MSI build failed.
    exit /b 1
)

echo.
echo MSI created successfully:
echo   %OUTPUT_DIR%\NERC_TPL_CA_%VERSION%_x64.msi
exit /b 0
