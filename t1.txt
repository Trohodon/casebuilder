"""Per-case text logging for PowerWorld analysis runs."""

from __future__ import annotations

from datetime import datetime
from pathlib import Path
from typing import Callable, TextIO


def case_log_path(output_dir: str | Path, case_path: str | Path) -> Path:
    """Return the legacy-compatible logfile name for one input case."""
    return Path(output_dir) / f"NERC TPL CA LOGFILE_{Path(case_path).name}.txt"


class CaseSessionLog:
    """A fresh, immediately flushed logfile for one case analysis."""

    def __init__(
        self,
        output_dir: str | Path,
        case_path: str | Path,
        *,
        now: Callable[[], datetime] = datetime.now,
    ) -> None:
        self.path = case_log_path(output_dir, case_path)
        self._now = now
        self._stream: TextIO | None = None

    def __enter__(self) -> CaseSessionLog:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        # Opening with "w" intentionally clears the previous run's logfile.
        self._stream = self.path.open("w", encoding="utf-8")
        return self

    def write(self, message: object) -> str:
        if self._stream is None:
            raise RuntimeError("The case session log is not open.")
        timestamp = self._now().strftime("%Y-%m-%d %H:%M:%S")
        lines = str(message).splitlines() or [""]
        formatted = "\n".join(f"{timestamp}: {line}" for line in lines)
        self._stream.write(formatted + "\n")
        self._stream.flush()
        return formatted

    def __exit__(self, _exc_type, _exc, _traceback) -> None:
        if self._stream is not None:
            self._stream.close()
            self._stream = None
